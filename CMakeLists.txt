cmake_minimum_required(VERSION 3.16)
project(autojam CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT CMAKE_BUILD_TYPE)
  set(CMAKE_BUILD_TYPE Release)
endif()
set(CMAKE_CXX_FLAGS_RELEASE "-O2")
find_package(PkgConfig REQUIRED)
pkg_check_modules(SDL2 REQUIRED sdl2)
pkg_check_modules(SDL2_TTF REQUIRED SDL2_ttf)
pkg_check_modules(PULSE REQUIRED libpulse-simple)
file(GLOB SRC ${CMAKE_SOURCE_DIR}/src/*.cpp)
add_executable(autojam ${SRC})
target_include_directories(autojam PRIVATE ${SDL2_INCLUDE_DIRS} ${SDL2_TTF_INCLUDE_DIRS} ${PULSE_INCLUDE_DIRS})
target_link_libraries(autojam ${SDL2_LIBRARIES} ${SDL2_TTF_LIBRARIES} ${PULSE_LIBRARIES} m pthread)
target_compile_options(autojam PRIVATE -Wall -Wextra -Wno-unused-parameter -Wno-missing-field-initializers -Wno-misleading-indentation)
